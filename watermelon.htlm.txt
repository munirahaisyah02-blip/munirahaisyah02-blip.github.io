<script>
    const ctx = document.getElementById('brixChart').getContext('2d');
    
    // x = Days into the ripening phase (0 to 30)
    const days = Array.from({length: 31}, (_, i) => i);
    // f(x) = Brix value
    const brixData = days.map(x => (-0.01 * Math.pow(x, 2) + 0.6 * x + 3).toFixed(2));

    const myChart = new Chart(ctx, {
        type: 'line',
        data: {
            labels: days,
            datasets: [{
                label: 'Sweetness (Degrees Brix)',
                data: brixData,
                borderColor: '#d63031',
                backgroundColor: 'rgba(255, 118, 117, 0.2)',
                pointBackgroundColor: '#d63031',
                pointHoverRadius: 8,
                fill: true,
                tension: 0.3
            }]
        },
        options: {
            plugins: { legend: { display: false } },
            scales: {
                x: { title: { display: true, text: 'Days of Ripening Phase' } },
                y: { 
                    min: 0, 
                    max: 15,
                    title: { display: true, text: 'Brix Content (°Bx)' } 
                }
            },
            onClick: (evt) => {
                const points = myChart.getElementsAtEventForMode(evt, 'nearest', { intersect: true }, true);
                if (points.length) {
                    const i = points[0].index;
                    const day = days[i];
                    const brix = brixData[i];
                    
                    // Derivative f'(x) = -0.02x + 0.6
                    const rate = (-0.02 * day + 0.6).toFixed(3);
                    
                    displayAnalysis(day, brix, rate);
                }
            }
        }
    });

    function displayAnalysis(day, brix, rate) {
        const res = document.getElementById('result');
        let comment = "";

        if (rate > 0.4) {
            comment = "<strong>Active Synthesis:</strong> Sugar is pumping into the fruit rapidly. Do not harvest yet; the flavor is still developing.";
        } else if (rate > 0.1) {
            comment = "<strong>Peak Sweetness Approaching:</strong> The accumulation is slowing down. Sugar levels are near maximum. Ideal for premium market harvest.";
        } else if (rate > 0) {
            comment = "<strong>Full Maturity:</strong> Minimal sugar gain remaining. Harvest immediately to ensure crisp texture before the flesh softens.";
        } else {
            comment = "<strong>Overripe/Senescence:</strong> Sugar levels are actually dropping due to metabolic breakdown. Quality is degrading.";
        }

        res.innerHTML = `
            <strong>Analysis - Day ${day}:</strong> Sugar Content is <strong>${brix}°Bx</strong>.<br>
            <strong>Derivative f'(${day}):</strong> ${rate} °Bx per day.<br>
            <em>Agro-Tech Note:</em> ${comment}
        `;
        res.style.background = "#55efc4"; // Flash green on update
        setTimeout(() => { res.style.background = "#fab1a0"; }, 500);
    }
</script>
