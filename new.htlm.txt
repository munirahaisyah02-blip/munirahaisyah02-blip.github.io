<script>
    const ctx = document.getElementById('integralChart').getContext('2d');
    const labels = Array.from({length: 101}, (_, i) => i);
    const dataPoints = labels.map(x => (100 * Math.exp(-0.02 * x)).toFixed(2));

    const chart = new Chart(ctx, {
        type: 'line',
        data: {
            labels: labels,
            datasets: [{
                label: 'Vitamin C Retention (%)',
                data: dataPoints,
                borderColor: '#2980b9',
                backgroundColor: 'rgba(41, 128, 185, 0.2)',
                fill: 'origin', // This visualizes the Area Under the Curve
                tension: 0.3,
                pointRadius: 0,
                pointHoverRadius: 5
            }]
        },
        options: {
            interaction: { mode: 'index', intersect: false },
            plugins: { 
                tooltip: { enabled: false },
                legend: { display: false }
            },
            scales: {
                y: { min: 0, max: 100, title: { display: true, text: 'Vitamin C %' } },
                x: { title: { display: true, text: 'Seconds in Blancher' } }
            },
            onClick: (e) => {
                const points = chart.getElementsAtEventForMode(e, 'index', { intersect: false }, true);
                if (points.length) {
                    const x = points[0].index;
                    calculateIntegral(x);
                }
            }
        }
    });

    function calculateIntegral(t) {
        const retention = parseFloat(dataPoints[t]);
        // Total Loss = Initial (100) - Current Retention
        const totalLoss = (100 - retention).toFixed(2);
        
        document.getElementById('stat-time').innerText = t;
        document.getElementById('stat-retention').innerText = retention + "%";
        document.getElementById('stat-loss').innerText = totalLoss + "%";
        
        // Visual feedback on the stat card
        const lossCard = document.getElementById('stat-loss').parentElement;
        lossCard.style.transform = "scale(1.05)";
        setTimeout(() => lossCard.style.transform = "scale(1)", 200);
    }
</script>
